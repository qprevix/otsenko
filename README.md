(function(){
var SEC=[{k:'m',n:'Мужчина',i:'♂'},{k:'f',n:'Девушка',i:'♀'}];
var db=null,uid=null,canWrite=false,isOwner=false,g='m',photo='';
var people=[],ratings=[],comments=[],sugg=[],open={},drafts={},editId=null,editDraft={};
var $=function(i){return document.getElementById(i);};
function esc(s){var d=document.createElement('div');d.textContent=s==null?'':String(s);return d.innerHTML;}
function norm(s){return String(s||'').toLowerCase().replace(/ё/g,'е').replace(/[^a-zа-я0-9]+/g,' ').trim();}
function plural(n){var m=n%100,k=n%10;if(m>10&&m<15)return 'оценок';if(k===1)return 'оценка';if(k>1&&k<5)return 'оценки';return 'оценок';}
function av(p,s){var c='av'+(s?' s':'');
  if(p.photo&&/^data:image\//.test(p.photo))return '<div class="'+c+'"><img src="'+esc(p.photo)+'" alt=""></div>';
  return '<div class="'+c+'">'+esc((p.name||'?').charAt(0).toUpperCase())+'</div>';}

function editForm(p){
  var d=editDraft;
  return '<div class="card"><div class="pick">'+av({name:d.name,photo:d.photo||p.photo},1)+'<label class="f">Заменить фото<input type="file" id="e_ph" accept="image/*"></label></div>'
   +'<select data-e="g"><option value="m"'+(d.g==='m'?' selected':'')+'>♂ Мужчина</option><option value="f"'+(d.g==='f'?' selected':'')+'>♀ Девушка</option></select>'
   +'<input type="text" data-e="name" maxlength="80" value="'+esc(d.name)+'"><textarea data-e="about" maxlength="200">'+esc(d.about)+'</textarea>'
   +'<div class="two"><button class="btn" data-esave="1">Сохранить</button><button class="btn sec" data-ecancel="1">Отмена</button></div></div>';
}
function render(){
  var st={};
  people.forEach(function(p){st[p.id]={s:0,n:0,m:0,c:0};});
  ratings.forEach(function(r){var x=st[r.personId];if(!x)return;x.s+=r.value;x.n++;if(r.userId===uid)x.m=r.value;});
  comments.forEach(function(c){var x=st[c.personId];if(x&&c.status==='ok')x.c++;});
  document.body.className=g==='f'?'f':'';
  $('tabs').innerHTML=SEC.map(function(c){return '<button data-tab="'+c.k+'"'+(c.k===g?' class="on"':'')+'>'+c.i+' '+c.n+'</button>';}).join('');
  var rows=people.filter(function(p){return p.g===g;}).map(function(p){var x=st[p.id];return {p:p,avg:x.n?x.s/x.n:0,n:x.n,m:x.m,c:x.c};});
  rows.sort(function(a,b){return b.avg-a.avg||b.n-a.n;});
  var tp=rows.filter(function(r){return r.n>0;}).slice(0,3),md=['🥇','🥈','🥉'];
  $('top').innerHTML=tp.length?'<h2>🏆 Топ по рейтингу</h2>'+tp.map(function(r,i){
    return '<div class="t'+(i===0?' first':'')+'"><div class="m">'+md[i]+'</div>'+av(r.p,1)+'<div class="info"><div class="name">'+esc(r.p.name)+'</div><div class="st">Средняя оценка: <b>'+r.avg.toFixed(1)+'</b> · проголосовало: '+r.n+'</div></div></div>';
  }).join('')+'<h2>Все люди</h2>':'';
  if(!rows.length)$('list').innerHTML='<div class="empty">В этом разделе пока никого нет.</div>';
  else $('list').innerHTML=rows.map(function(r){
    var id=r.p.id,btns='';
    if(isOwner&&editId===id)return editForm(r.p);
    if(canWrite)for(var k=1;k<=10;k++)btns+='<button data-p="'+esc(id)+'" data-v="'+k+'"'+(r.m===k?' class="mine"':'')+'>'+k+'</button>';
    var cm='';
    if(open[id]){
      var cl=comments.filter(function(c){return c.personId===id&&(c.status==='ok'||c.userId===uid);}).sort(function(a,b){return b.created-a.created;});
      cm='<div class="cm">'+(cl.length?cl.map(function(c){return '<div class="cmi">'+esc(c.text)+'<small>'+(c.status==='ok'?'Аноним':'Ждёт проверки')+'</small></div>';}).join(''):'<div class="note">Комментариев пока нет</div>')
        +(canWrite?'<textarea data-draft="'+esc(id)+'" maxlength="300" placeholder="Ваш комментарий">'+esc(drafts[id]||'')+'</textarea><button class="btn" data-send="'+esc(id)+'">Отправить комментарий</button>':'')+'</div>';
    }
    return '<div class="card"><div class="row">'+av(r.p)+'<div class="info"><div class="name">'+esc(r.p.name)+'</div>'+(r.p.about?'<div class="about">'+esc(r.p.about)+'</div>':'')
      +'</div><div class="score"><b>'+(r.n?r.avg.toFixed(1):'—')+'</b><span>проголосовало: '+r.n+'</span></div></div>'
      +(canWrite?'<div class="rate">'+btns+'</div>':'')
      +'<button class="lnk" data-cm="'+esc(id)+'">💬 Комментарии ('+r.c+')</button>'
      +(isOwner?'<button class="lnk g" data-edit="'+esc(id)+'">Редактировать</button><button class="lnk g" data-del="'+esc(id)+'">Удалить</button>':'')+cm+'</div>';
  }).join('');
  var nm=function(id){var x=people.filter(function(i){return i.id===id;})[0];return x?x.name:'—';};
  var pc=comments.filter(function(c){return c.status==='pending';});
  var un=people.filter(function(p){return p.g!=='m'&&p.g!=='f';});
  $('adm').innerHTML=isOwner?
    (un.length?'<h2>Без раздела ('+un.length+')</h2>'+un.map(function(p){return '<div class="card"><div class="row">'+av(p,1)+'<div class="info"><div class="name">'+esc(p.name)+'</div></div></div><div class="two"><button class="btn sec" data-sg="m" data-id="'+esc(p.id)+'">♂ Мужчина</button><button class="btn sec" data-sg="f" data-id="'+esc(p.id)+'">♀ Девушка</button></div></div>';}).join(''):'')
    +(sugg.length?'<h2>Предложения ('+sugg.length+')</h2>'+sugg.map(function(s){return '<div class="card"><div class="row">'+av(s,1)+'<div class="info"><div class="name">'+(s.g==='f'?'♀':'♂')+' '+esc(s.name)+'</div>'+(s.about?'<div class="about">'+esc(s.about)+'</div>':'')+'</div></div><div class="two"><button class="btn" data-ok="'+esc(s.id)+'">Добавить</button><button class="btn sec" data-no="'+esc(s.id)+'">Отклонить</button></div></div>';}).join(''):'')
    +(pc.length?'<h2>Комментарии на проверке ('+pc.length+')</h2>'+pc.map(function(c){return '<div class="card"><div class="about">К: '+esc(nm(c.personId))+'</div><div class="cmi" style="margin-top:6px">'+esc(c.text)+'</div><div class="two"><button class="btn" data-cok="'+esc(c.id)+'">Опубликовать</button><button class="btn sec" data-cno="'+esc(c.id)+'">Удалить</button></div></div>';}).join(''):''):'';
}

document.addEventListener('input',function(e){var x=e.target.dataset;if(x.draft)drafts[x.draft]=e.target.value;if(x.e)editDraft[x.e]=e.target.value;});
document.addEventListener('change',function(e){
  if(e.target.id!=='e_ph')return;var f=e.target.files[0];if(!f)return;var fr=new FileReader();
  fr.onload=function(){var im=new Image();im.onload=function(){var sc=Math.min(1,320/Math.max(im.width,im.height)),c=document.createElement('canvas');c.width=Math.round(im.width*sc);c.height=Math.round(im.height*sc);c.getContext('2d').drawImage(im,0,0,c.width,c.height);editDraft.photo=c.toDataURL('image/jpeg',0.8);render();};im.src=fr.result;};fr.readAsDataURL(f);
});
document.addEventListener('click',function(e){
  var t=e.target,d=t.dataset;
  if(d.tab){g=d.tab;render();return;}
  if(d.cm){open[d.cm]=!open[d.cm];render();return;}
  if(!db)return;
  if(d.v&&uid){db.doc('ratings/'+d.p+'_'+uid).set({personId:d.p,userId:uid,value:+d.v}).catch(function(){$('note').textContent='Не удалось сохранить оценку.';});}
  else if(d.send&&uid){
    var tx=(drafts[d.send]||'').trim();if(!tx)return;
    db.collection('comments').add({personId:d.send,userId:uid,text:tx.slice(0,300),status:isOwner?'ok':'pending',created:Date.now()})
      .then(function(){drafts[d.send]='';if(!isOwner)$('note').textContent='Комментарий отправлен на проверку.';});
  }
  else if(d.del){
    if(d.sure!=='1'){t.dataset.sure='1';t.textContent='Нажмите ещё раз';setTimeout(function(){t.dataset.sure='';t.textContent='Удалить';},4000);return;}
    var id=d.del;
    Promise.resolve(db.doc('people/'+id).delete()).catch(function(){$('note').textContent='Не удалось удалить.';});
    ratings.filter(function(r){return r.personId===id;}).forEach(function(r){db.doc('ratings/'+r.id).delete();});
    comments.filter(function(c){return c.personId===id;}).forEach(function(c){db.doc('comments/'+c.id).delete();});
  }
  else if(d.sg){
    var p=people.filter(function(x){return x.id===d.id;})[0];if(!p)return;
    db.doc('people/'+p.id).set({name:p.name,about:p.about||'',photo:p.photo||'',g:d.sg,created:p.created||Date.now()});
  }
  else if(d.ok){
    var s=sugg.filter(function(x){return x.id===d.ok;})[0];if(!s)return;
    var dup=people.some(function(i){return i.g===s.g&&norm(i.name)===norm(s.name);});
    var pr=dup?Promise.resolve():db.collection('people').add({name:s.name,about:s.about||'',photo:s.photo||'',g:s.g,created:Date.now()});
    pr.then(function(){return db.doc('suggestions/'+s.id).delete();});
  }
  else if(d.no){db.doc('suggestions/'+d.no).delete();}
  else if(d.cok){var c=comments.filter(function(x){return x.id===d.cok;})[0];if(c)db.doc('comments/'+c.id).set({personId:c.personId,userId:c.userId,text:c.text,status:'ok',created:c.created});}
  else if(d.cno){db.doc('comments/'+d.cno).delete();}
  else if(d.edit){var q=people.filter(function(x){return x.id===d.edit;})[0];if(q){editId=q.id;editDraft={name:q.name,about:q.about||'',g:(q.g==='f'?'f':'m'),photo:''};render();}}
  else if(d.ecancel){editId=null;render();}
  else if(d.esave){
    var o=people.filter(function(x){return x.id===editId;})[0];if(!o)return;
    var nn=(editDraft.name||'').trim();if(!nn)return;
    if(people.some(function(x){return x.id!==editId&&x.g===editDraft.g&&norm(x.name)===norm(nn);})){$('note').textContent='Такой человек уже есть в этом разделе.';return;}
    db.doc('people/'+editId).set({name:nn,about:(editDraft.about||'').trim(),photo:editDraft.photo||o.photo||'',g:editDraft.g,created:o.created||Date.now()}).then(function(){editId=null;render();});
  }
});

});

$('addBtn').addEventListener('click',function(){
  var name=$('nm').value.trim(),gd=$('gd').value;if(!name||!db)return;
  if(!$('cs').checked){$('addMsg').textContent='Подтвердите согласие человека.';return;}
  var key=norm(name);
  if(people.some(function(i){return i.g===gd&&norm(i.name)===key;})){$('addMsg').textContent='Такой человек уже есть в списке.';return;}
  if(sugg.some(function(s){return s.g===gd&&norm(s.name)===key;})){$('addMsg').textContent='Этого человека уже предложили, он ждёт проверки.';return;}
  var body={name:name,about:$('ab').value.trim(),photo:photo,g:gd,addedBy:uid,created:Date.now()};
  var p=isOwner?db.collection('people').add(body):db.collection('suggestions').add(body);
  p.then(function(){$('nm').value='';$('ab').value='';photo='';$('cs').checked=false;$('prev').textContent='+';$('addMsg').textContent=isOwner?'Добавлено!':'Спасибо! Предложение отправлено на проверку.';})
   .catch(function(){$('addMsg').textContent='Не удалось отправить.';});
});

async function init(){
  try{
    if(!window.claude||!claude.use){$('note').textContent='Не удалось подключиться к хранилищу.';return;}
    db=await claude.use('db');
    var user=await claude.use('user');
    if(!db){$('list').innerHTML='<div class="empty">Войдите в аккаунт, чтобы видеть список.</div>';return;}
    if(user){
      try{uid=await user.id();}catch(e){}
      isOwner=user.isOwner();
      var cw=user.can('data.write');
      canWrite=!!uid&&cw!==false;
    }
    if(canWrite){$('addBox').hidden=false;$('addH').hidden=false;if(isOwner){$('addBtn').textContent='Добавить в список';$('addH').textContent='Добавить человека';}}
    else $('note').textContent='Просмотр без права голосовать. Нужен доступ «Участник».';
    var map=function(s){return s.docs.map(function(d){var x=d.data();x.id=d.id;return x;});};
    db.collection('people').onSnapshot(function(s){people=map(s);render();},function(){$('list').innerHTML='<div class="empty">Не удалось загрузить список.</div>';});
    db.collection('ratings').onSnapshot(function(s){ratings=map(s);render();});
    db.collection('comments').onSnapshot(function(s){comments=map(s);render();});
    db.collection('suggestions').onSnapshot(function(s){sugg=map(s);render();});
  }catch(e){$('list').innerHTML='<div class="empty">Ошибка загрузки.</div>';}
}
render();init();
})();
